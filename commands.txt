# Starterm commands file
# ======================
# Format:
#   command                        -> shown as-is
#   command :: description         -> description shown in the dropdown
#
# Notes:
#   - Lines starting with '#' are ignored.
#   - Blank lines are ignored.
#   - Use "::" as the separator so shell pipes (|) inside the command
#     are preserved (e.g. "ps aux | head -20").
#   - Commands with destructive potential are filtered by the app.

echo $SHELL

- #### -- Calendar & time -- #####

cal                                       :: Show this month's calendar
date                                      :: Show current date and time
date -u +'%Y-%m-%d %H:%M:%S %Z'           :: UTC date/time in ISO-ish format
date +'%Y-%m-%d %H:%M:%S %Z'              :: Local date/time in ISO-ish format
date -u +'%Y-%m-%d %H:%M:%S %z'           :: UTC date/time with numeric offset
date +'%Y-%m-%d %H:%M:%S %z'              :: Local date/time with numeric offset
echo = ; cal -3 ; echo = ; date           :: Combined calendar + date banner
